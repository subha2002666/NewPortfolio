<script>
/* ---------------------------
   Canvas particles + cursor trail
   --------------------------- */
(() => {
  const canvas = document.getElementById('bgCanvas');
  const ctx = canvas.getContext('2d');
  let w = canvas.width = innerWidth;
  let h = canvas.height = innerHeight;
  const particles = [];
  const cursor = {x:-9999,y:-9999};

  function rand(min,max){ return Math.random()*(max-min)+min }

  // create drifting particles
  for(let i=0;i<80;i++){
    particles.push({
      x: rand(0,w),
      y: rand(0,h),
      vx: rand(-0.2,0.2),
      vy: rand(-0.1,0.4),
      r: rand(0.6,2.6),
      life: rand(60,240),
      hue: rand(0,360),
      alpha: rand(0.06,0.18)
    });
  }

  function resize(){ w=canvas.width=innerWidth; h=canvas.height=innerHeight; }
  addEventListener('resize', resize);

  addEventListener('mousemove', e => { cursor.x=e.clientX; cursor.y=e.clientY; });
  addEventListener('mouseleave', ()=>{ cursor.x=-9999; cursor.y=-9999; });

  function draw(){
    ctx.clearRect(0,0,w,h);
    // subtle vignette
    const g = ctx.createLinearGradient(0,0,w,h);
    g.addColorStop(0,'rgba(0,0,0,0)');
    g.addColorStop(1,'rgba(0,0,0,0.25)');
    ctx.fillStyle = g;
    ctx.fillRect(0,0,w,h);

    // draw particles
    particles.forEach((p,i)=>{
      p.x += p.vx + Math.sin((p.y+p.x)/120)*0.1;
      p.y += p.vy;
      p.life--;
      if(p.y>h+20 || p.x<-40 || p.x>w+40 || p.life<0){
        p.x = rand(0,w); p.y = -10; p.life = rand(60,240);
      }
      // glow
      ctx.beginPath();
      ctx.fillStyle = `rgba(255,26,26,${p.alpha})`;
      ctx.shadowBlur = 12;
      ctx.shadowColor = 'rgba(255,26,26,0.12)';
      ctx.arc(p.x,p.y,p.r,0,Math.PI*2);
      ctx.fill();
      ctx.shadowBlur = 0;
    });

    // cursor trail
    if(cursor.x>-9000){
      for(let i=0;i<6;i++){
        const t = Math.random()*18+6;
        ctx.beginPath();
        ctx.fillStyle = `rgba(255,26,26,${0.08 + Math.random()*0.12})`;
        ctx.arc(cursor.x + Math.sin(Date.now()/200 + i)*12, cursor.y + Math.cos(Date.now()/180 + i)*12, t/12, 0, Math.PI*2);
        ctx.fill();
      }
    }

    requestAnimationFrame(draw);
  }
  draw();
})();

    dot.className = 'dot-btn';
    dot.addEventListener('click', ()=> setIndex(idx));
    dots.appendChild(dot);
  });

  const cards = Array.from(document.querySelectorAll('.card3d'));
  let index = 0;
  function layout(){
    const mid = Math.floor(cards.length/2);
    cards.forEach((c,i)=>{
      const offset = i - index;
      const abs = Math.abs(offset);
      const z = 300 - abs*120;
      const x = offset * 220;
      const rot = offset * -12;
      c.style.opacity = abs>3 ? 0 : 1;
      c.style.transform = `translate3d(${x}px, -50%, ${-abs*60}px) rotateY(${rot}deg)`;
      c.style.left = '50%';
      c.style.top = '50%';
      c.style.zIndex = 100 - abs;
    });
    // dots
    dots.querySelectorAll('.dot-btn').forEach((d,i)=> d.classList.toggle('active', i===index));
  }
  function setIndex(i){
    index = (i + cards.length) % cards.length;
    layout();
  }
  // initial
  setIndex(0);
  // auto rotate
  let auto = setInterval(()=> setIndex((index+1)%cards.length), 4200);
  carousel.addEventListener('mouseenter', ()=> clearInterval(auto));
  carousel.addEventListener('mouseleave', ()=> auto = setInterval(()=> setIndex((index+1)%cards.length), 4200));
})();
